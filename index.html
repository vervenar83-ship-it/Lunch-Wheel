let menus = []; const canvas = document.getElementById('roulette'); const ctx = canvas.getContext('2d'); // 페이지 로드 시 URL에 메뉴 데이터가 있으면 불러오기 window.onload = () => { const params = new URLSearchParams(window.location.search); if (params.has('data')) { try { menus = JSON.parse(atob(params.get('data'))); draw(); } catch (e) { console.error("데이터를 불러오는 중 오류가 발생했습니다."); } } }; function addMenu() { const input = document.getElementById('menuInput'); if (input.value) { menus.push(input.value); input.value = ''; draw(); } } function draw() { ctx.clearRect(0, 0, 400, 400); if (menus.length === 0) return; const arc = Math.PI * 2 / menus.length; menus.forEach((menu, i) => { const angle = i * arc; // 부채꼴 그리기 ctx.beginPath(); ctx.fillStyle = `hsl(${i * 360 / menus.length}, 70%, 70%)`; ctx.moveTo(200, 200); ctx.arc(200, 200, 200, angle, angle + arc); ctx.fill(); ctx.stroke(); // 텍스트 배치 ctx.save(); ctx.translate(200, 200); ctx.rotate(angle + arc / 2); ctx.fillStyle = '#000'; ctx.font = '16px Arial'; ctx.fillText(menu, 100, 10); ctx.restore(); }); } function spin() { // 3초간 회전 후 멈춤 const rotation = Math.random() * 360 + 720; canvas.style.transition = 'transform 3s ease-out'; canvas.style.transform = `rotate(${rotation}deg)`; } function resetRoulette() { menus = []; canvas.style.transition = 'none'; canvas.style.transform = 'rotate(0deg)'; window.history.pushState({}, '', window.location.pathname); draw(); } function shareLink() { const data = btoa(JSON.stringify(menus)); const url = window.location.origin + window.location.pathname + '?data=' + data; navigator.clipboard.writeText(url).then(() => alert('링크가 복사되었습니다.')); }
