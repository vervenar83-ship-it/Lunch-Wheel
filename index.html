function draw() { ctx.clearRect(0, 0, 400, 400); // 1. 룰렛의 기본 테두리(원형)를 항상 먼저 그리기 ctx.beginPath(); ctx.arc(200, 200, 200, 0, Math.PI * 2); ctx.stroke(); const keys = Object.keys(menuCounts); if (keys.length === 0) return; const total = Object.values(menuCounts).reduce((a, b) => a + b, 0); let currentAngle = 0; // 2. 메뉴 영역 그리기 keys.forEach((menu, i) => { const sliceAngle = (menuCounts[menu] / total) * Math.PI * 2; ctx.beginPath(); ctx.moveTo(200, 200); ctx.fillStyle = `hsl(${i * 360 / keys.length}, 70%, 70%)`; ctx.arc(200, 200, 200, currentAngle, currentAngle + sliceAngle); ctx.fill(); ctx.stroke(); // 부채꼴 선 ctx.save(); ctx.translate(200, 200); ctx.rotate(currentAngle + sliceAngle / 2); ctx.fillStyle = '#000'; ctx.font = '14px Arial'; ctx.fillText(`${menu} (${menuCounts[menu]})`, 80, 5); ctx.restore(); currentAngle += sliceAngle; }); }
